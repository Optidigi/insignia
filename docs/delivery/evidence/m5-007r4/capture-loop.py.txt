"""One reserved official command with private output; no automatic retry."""
import datetime
import json
import os
from pathlib import Path
import signal
import subprocess
import sys
import time
from sanitizer import project

ROOT = Path(__file__).parent
REGISTER = ROOT / "register.json"


def atomic_json(path, value):
    temporary = path.with_name(path.name + ".pending")
    fd = os.open(temporary, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o600)
    with os.fdopen(fd, "w") as stream:
        json.dump(value, stream, indent=2)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())
    os.replace(temporary, path)


slot_number = int(sys.argv[1])
state = json.loads(REGISTER.read_text())
controls = json.loads((ROOT / "sanitized/sanitizer-controls.json").read_text())
assert controls["status"] == "PASS" and controls["count"] >= 5
assert state["consumed"] == slot_number - 1
assert slot_number in (1, 2, 3)
if slot_number == 1:
    assert state["status"] == "PREPARING_CONTROLS"
else:
    assert state["status"] == f"SLOT_{slot_number - 1}_SUCCEEDED"
    assert state["slots"][-1]["expectedContextUnambiguous"]
context = ROOT / "version-context"
capture = ROOT / "capture"
versions = ["app", "versions", "list", "--client-id", "1443cf6d03d39edae7c101a943c5c684", "--path", str(context), "--json"]
operation = versions if slot_number in (1, 3) else ["app", "config", "link", "--client-id", "1443cf6d03d39edae7c101a943c5c684", "--path", str(capture), "--file-name", "m5-007r2"]
if slot_number == 2:
    assert not list(capture.iterdir())
argv = [state["node"], state["entrypoint"], *operation]
cwd = context if slot_number in (1, 3) else capture
environment = {key: os.environ[key] for key in ("HOME", "USER", "LOGNAME")}
assert environment == {"HOME": "/home/serveradmin", "USER": "serveradmin", "LOGNAME": "serveradmin"}
environment.update(PATH="/home/serveradmin/.local/opt/node-v24.21.0-linux-x64/bin:/usr/bin:/bin", CI="1", SHOPIFY_CLI_NO_ANALYTICS="1", LANG="C.UTF-8", TZ="UTC", TERM="dumb", NO_COLOR="1", FORCE_COLOR="0")
assert "SHOPIFY_CLI_FORCE_AUTO_UPGRADE" not in environment
stamp = datetime.datetime.now(datetime.timezone.utc).isoformat()
reservation = {
    "slot": slot_number, "argv": argv, "cwd": str(cwd), "reservedAt": stamp,
    "environmentNamesSet": sorted(environment),
    "knownPublicValues": {"CI": "1", "SHOPIFY_CLI_NO_ANALYTICS": "1"},
    "environmentNamesExplicitlyUnset": ["SHOPIFY_CLI_FORCE_AUTO_UPGRADE", "SHOPIFY_FLAG_*", "alternate token/auth/proxy overrides", "NODE_OPTIONS", "NODE_PATH", "debug/plugin/Oclif overrides", "all non-allowlisted inherited names"],
    "stdin": "CLOSED", "tty": False, "deadlineSeconds": 120, "status": "RESERVED",
}
state["slots"].append(reservation)
state["consumed"] = slot_number
state["status"] = "RUNNING"
atomic_json(REGISTER, state)
stdout_path = ROOT / "raw" / f"command-{slot_number}.stdout"
stderr_path = ROOT / "raw" / f"command-{slot_number}.stderr"
assert stdout_path.stat().st_mode & 0o777 == 0o600
assert stderr_path.stat().st_mode & 0o777 == 0o600
assert stdout_path.stat().st_size == stderr_path.stat().st_size == 0
start = time.monotonic()
timed_out = False
with stdout_path.open("wb") as stdout, stderr_path.open("wb") as stderr:
    process = subprocess.Popen(argv, cwd=cwd, env=environment, stdin=subprocess.DEVNULL, stdout=stdout, stderr=stderr, start_new_session=True)
    reservation["ownedProcessGroup"] = process.pid
    reservation["status"] = "RUNNING"
    atomic_json(REGISTER, state)
    try:
        process.wait(timeout=max(0.1, 120 - (time.monotonic() - start)))
    except subprocess.TimeoutExpired:
        timed_out = True
        os.killpg(process.pid, signal.SIGKILL)
        process.wait()
    stdout.flush()
    stderr.flush()
    os.fsync(stdout.fileno())
    os.fsync(stderr.fileno())
try:
    os.killpg(process.pid, 0)
    descendants_remained = True
    os.killpg(process.pid, signal.SIGKILL)
except ProcessLookupError:
    descendants_remained = False
duration = time.monotonic() - start
stdout = stdout_path.read_text(errors="replace")
stderr = stderr_path.read_text(errors="replace")
sanitized = project(stdout, stderr)
accepted = False
if process.returncode == 0 and not timed_out and not descendants_remained and not sanitized["updaterOrInstallAttemptObservedInOutput"]:
    if slot_number in (1, 3):
        try:
            versions_data = json.loads(stdout)
            assert isinstance(versions_data, list)
            facts = [{key: entry[key] for key in ("versionTag", "versionId", "status", "createdAt") if key in entry} for entry in versions_data]
            active = [entry for entry in facts if entry.get("status") == "active"]
            assert len(active) == 1 and str(active[0].get("versionId")) in ("1146748534785", "gid://shopify/Version/1146748534785") and active[0].get("versionTag") == "insignia-1"
            sanitized["publicVersions"] = facts
            accepted = True
        except (ValueError, TypeError, AssertionError):
            sanitized["classification"] = "VERSION_OR_OUTPUT_MISMATCH"
    else:
        import tomllib
        linked = capture / "shopify.app.m5-007r2.toml"
        try:
            configuration = tomllib.loads(linked.read_text())
            accepted = configuration.get("client_id") == "1443cf6d03d39edae7c101a943c5c684"
        except (OSError, ValueError):
            accepted = False
        if not accepted:
            sanitized["classification"] = "CAPTURE_CONTEXT_MISMATCH"
if timed_out:
    sanitized["classification"] = "DEADLINE_EXCEEDED"
elif descendants_remained:
    sanitized["classification"] = "OWNED_DESCENDANT_SAFETY_STOP"
elif sanitized["updaterOrInstallAttemptObservedInOutput"]:
    sanitized["classification"] = "UPDATER_OR_INSTALL_SAFETY_STOP"
reservation.update(
    status="SUCCEEDED" if accepted else "FAILED_STOP",
    exitCode=process.returncode if process.returncode >= 0 else None,
    signal=(-process.returncode) if process.returncode < 0 else None,
    elapsedSeconds=round(duration, 3), timeout=timed_out,
    ownedDescendantsRemained=descendants_remained,
    expectedContextUnambiguous=accepted,
    classification=sanitized["classification"],
    finishedAt=datetime.datetime.now(datetime.timezone.utc).isoformat(),
    rawOutputDisposition="RETAINED_PRIVATE_MODE_0600_NOT_FOR_EXPORT",
    rawStdoutBytes=stdout_path.stat().st_size, rawStderrBytes=stderr_path.stat().st_size,
)
state["status"] = f"SLOT_{slot_number}_SUCCEEDED" if accepted else "STOPPED"
state["lockedSlots"] = [] if accepted and slot_number == 3 else list(range(slot_number + 1, 4))
atomic_json(ROOT / "sanitized" / f"command-{slot_number}.json", {"result": reservation, "diagnostic": sanitized})
atomic_json(REGISTER, state)
print(json.dumps({"slot": slot_number, "exitCode": reservation["exitCode"], "signal": reservation["signal"], "elapsedSeconds": reservation["elapsedSeconds"], "status": state["status"], "sanitizedDiagnostic": sanitized}))
